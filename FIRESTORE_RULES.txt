// Proyecto: barberia-taccu
//
// IMPORTANTE: este archivo es una copia de lo que hay que PUBLICAR en la
// consola. Cambiarlo acá no cambia nada en Firestore: hay que copiar y
// pegar en Firebase Console > Firestore Database > Reglas y publicar.
//
// Esta versión parte de la que escribiste el 13/10 y le vuelve a cerrar los
// agujeros que la auditoría había cerrado. Lo que ya estaba bien en tu versión
// se deja tal cual: la validación de reseñas para que los clientes puedan
// dejar reseña, la de mensajes, las ramas de occupied/espera/activos.
//
// Lo que cambió respecto de tu versión, y por qué:
//
//   1. config: escribir pasa a ser SÓLO del dueño. Con `isBarberoOrOwner()`
//      cualquier barbero podía, desde la consola del navegador, reescribir
//      precios, horarios o el reparto de ganancias.
//   2. config/finanzas: deja de ser legible con sólo estar registrado. Lleva
//      los porcentajes y los uid del equipo, y con `isSignedIn()` cualquier
//      cuenta creada lo leía.
//   3. config/barberos: regla aparte para que un barbero, en el caso de que
//      haga falta, sólo pueda tocar SU PROPIA entrada de la lista.
//   4. users: el tope del alta pasa de 20 a 50 puntos.
//   5. staff: un barbero ya NO puede tocar su campo `pd` (el porcentaje que se
//      queda el dueño). Con `pd` permitido, `updateDoc(doc(db,"staff",miUid),
//      {pd: 0})` le dejaba quedarse con el 100% de cada corte.
//   6. turnos: el alta valida el formato de la fecha y la hora; el cliente no
//      puede marcar su turno como "hecho", ponerle el precio, cambiar el barbero
//      ni borrar uno ya cobrado; y borrar un turno es sólo del equipo.
//   7. fotoBarbero: colección nueva para la foto de cada barbero.
//   8. ventas y gastos: un barbero sólo toca los suyos, no los del dueño.
//
// OJO: rules_version SÓLO admite '1' y '2'. No hay una versión 3: si ponés 3, la
// consola y los linters la rechazan ("Unsupported rules version 3"). El "número de
// versión" de este archivo es el del comentario de arriba, no esta línea.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() { return request.auth != null; }
    /* ===========================================================================
  CAMBIAR QUIEN ES EL DUEÑO

  El dueño se cambia acá, en ESTA línea, y en el archivo src/config.js
  (las constantes ADMIN_UID y OWNER_UID). Son los dos únicos lugares: este
  decide qué puede escribir y cuál es el dueño para el CLIENTE (la web), y
  el del código decide qué se ve en el panel.

  Hay que cambiar los dos JUNTOS y republicar estas reglas en la consola de
  Firebase. Si se cambia sólo uno de los dos, pasa una de estas dos cosas:

    - se cambia acá y no en el código: la persona nueva puede escribir en la
      base desde la consola, pero la web le muestra el panel de cliente y no
      aparece ni el nombre del barbero ni las fotos que se publican.
    - se cambia en el código y no acá: la persona nueva ve todo el panel de
      dueño, y cada guardar le rebota con permission-denied sin explicación.

  tests/owner.js compara los dos UID y falla si se separan, así que el error
  se ve antes de publicar.

  OJO: el barbero puede ser el dueño. Eso NO se decide acá ni en el config:
  se marca desde el panel, con "Yo también soy el barbero" en el menú de la
  foto. Es una bandera en la lista de barberos, no un permiso.
  =========================================================================== */
function isOwner() { return isSignedIn() && request.auth.uid == 'LQ7IkEfxEKNXajx4SEs378n390i2'; }
    function isStaff() { return isSignedIn() && exists(/databases/$(database)/documents/staff/$(request.auth.uid)); }
    function isBarberoOrOwner() { return isOwner() || isStaff(); }

    match /config/{docId} {
      // La landing lee sin sesión: servicios/precios (servicios), equipo
      // (barberos), horarios (horario) y las reglas de puntos (reglas).
      //
      // "finanzas" sí pide sesión: lleva los porcentajes de ganancia y los
      // uid de cada barbero. Con "allow read: if true" para todo /config esos
      // datos quedaban públicos en la web, y con "isSignedIn()" tampoco
      // alcanzaba: con sólo registrarse, cualquiera leía cuánto se lleva el
      // dueño de cada corte. Ahora hace falta ser del equipo.
      allow read: if docId != 'finanzas' || isStaff() || isOwner();

      // SÓLO el dueño escribe la configuración.
      //
      // Antes (y en tu versión) era `isBarberoOrOwner()`, o sea que cualquier
      // barbero podía, desde la consola del navegador:
      //
      //     setDoc(doc(db, "config", "finanzas"), { pd: 100 })
      //     setDoc(doc(db, "config", "horario"), { a: 540, c: 1380 })
      //
      // y a partir de ese momento se quedaba con el 100% de cada corte o
      // abría la barbería a la medianoche. La UI nunca habilitaba eso; el
      // problema es que las reglas son las que tienen que frenarlo.
      allow write: if isOwner();
    }

    // La lista pública de equipo la administra SOLO el dueño: él decide qué
    // barberos están y en qué orden, y puede quitar a uno.
    //
    // Antes esta colección tenía además una rama para que cada barbero
    // escribiera SU entrada de la lista, con esta condición:
    //
    //     request.resource.data.l.where(b, b.id != request.auth.uid)
    //          == resource.data.l.where(b, b.id != request.auth.uid)
    //
    // O sea: "todo lo que NO es mío tiene que quedar igual". El razonamiento
    // es correcto, pero en producción NUNCA dio true. Comprobado contra las
    // reglas de Firebase con la sesión real del barbero, escribiendo la lista
    // sin modificarla (idempotente, dato intacto): HTTP 403 "Missing or
    // insufficient permissions". Con el dueño el mismo pedido daba 200, así que
    // no era el payload: era la rama del barbero.
    //
    // El efecto era que "Editar perfil" fallaba siempre: guardarPerfil() hace
    // Promise.all entre config/barberos y staff/{uid}, y el primero rechazado
    // tumbaba el guardado entero.
    //
    // Por eso el perfil del barbero ya no vive acá: vive en su propio
    // documento, fotoBarbero/{uid} (abajo), que es una regla mucho más simple
    // y que además funciona.
    match /config/barberos {
      allow write: if isOwner();
    }

    match /galeria/{docId} {
      allow read: if true;
      allow create, update, delete: if isBarberoOrOwner();
    }

    // Reseñas. Se leen sin sesión (el carrusel de la web las muestra).
    //
    // IMPORTANTE: los clientes SÍ dejan dejar reseña desde su panel. Con
    // "create: if isBarberoOrOwner()" el write se denegaba siempre, porque
    // un cliente no es barbero ni dueño, y la reseña no se guardaba nunca.
    // Ahora cada cliente escribe la suya y no puede tocar la de otro; los
    // textos van acotados para que el campo no sirva de buzón.
    //
    // Lo que las reglas NO pueden impedir es que alguien se haga cuentas
    // y publique varias reseñas: eso requiere un contador por usuario, que
    // Firestore no permite sin backend. Si más adelante molesta, la
    // solución es un Cloud Function.
    match /resenas/{docId} {
      allow read: if true;
      allow create: if isBarberoOrOwner()
                    || (isSignedIn()
                        && request.resource.data.uid == request.auth.uid
                        && request.resource.data.keys().hasAll(['uid', 'nm', 'texto', 'estrellas'])
                        && request.resource.data.estrellas is int
                        && request.resource.data.estrellas >= 1
                        && request.resource.data.estrellas <= 5
                        && request.resource.data.nm is string
                        && request.resource.data.nm.size() <= 60
                        && request.resource.data.texto is string
                        && request.resource.data.texto.size() >= 1
                        && request.resource.data.texto.size() <= 600);
      allow update, delete: if isBarberoOrOwner()
                           || (isSignedIn() && resource.data.uid == request.auth.uid);
    }

    // ---------- usuarios ----------
    match /users/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isStaff() || isOwner());

      // Alta del perfil. Se acota qué se puede escribir y se clavan los
      // puntos entre 0 y 50 (los de bienvenida, o 0 si el cliente se da
      // de baja). Antes, con "create: if isSignedIn() && uid == uid",
      // cualquiera creaba su propio documento con el saldo que quisiera.
      //
      // El 20 era el tope de la versión anterior; el dueño pidió poder dar
      // hasta 50 puntos de bienvenida. OJO: los dos topes están atados. Si
      // acá va 50 y en el panel va 20, el dueño escribe 50, la base lo
      // rechaza y la cuenta queda creada en Auth sin documento.
      allow create: if isSignedIn()
                     && request.auth.uid == uid
                     && request.resource.data.keys().hasOnly(['nm', 'ph', 'pt', 'borrado'])
                     && request.resource.data.pt is int
                     && request.resource.data.pt >= 0
                     && request.resource.data.pt <= 50;

      // El cliente solo edita lo suyo: nombre, teléfono, foto y la baja.
      // Y los puntos SOLO pueden bajar: canjear un premio los descuenta, así
      // que "pt <= resource.data.pt" deja pasar el canje y frena que alguien
      // se los suba a mano desde la consola del navegador. Subirlos es cosa
      // de la barbería (cobro de un corte, venta manual), que entra por la
      // rama de isBarberoOrOwner() y no está limitada.
      allow update: if isBarberoOrOwner()
                    || (isSignedIn()
                        && request.auth.uid == uid
                        && request.resource.data.diff(resource.data).affectedKeys()
                             .hasOnly(['nm', 'ph', 'foto', 'borrado', 'pt'])
                        && request.resource.data.pt is number
                        && request.resource.data.pt <= resource.data.pt);

      allow delete: if isSignedIn() && (request.auth.uid == uid || isBarberoOrOwner());
    }

    // ---------- contacto ----------
    match /mensajes/{id} {
      // El formulario de la landing es público, así que el create sin sesión
      // es necesario. Pero "if true" lo convertía en un buzón abierto: se
      // validan los campos para que no entre basura ni documentos gigantic.
      // Lo que de verdad frena el spam es activar App Check en Firebase.
      allow create: if request.resource.data.nombre is string
                     && request.resource.data.nombre.size() <= 80
                     && request.resource.data.email is string
                     && request.resource.data.email.size() <= 120
                     && request.resource.data.mensaje is string
                     && request.resource.data.mensaje.size() >= 1
                     && request.resource.data.mensaje.size() <= 2000;
      allow read, update, delete: if isOwner();
    }

    match /staff/{uid} {
      allow read: if isSignedIn() && (request.auth.uid == uid || isStaff() || isOwner());
      // El dueño puede escribir cualquier ficha de barbero.
      allow create, delete: if isOwner();
      // Un barbero sólo puede tocar SU propia ficha, y sólo el campo "foto"
      // (y el nombre, que usa en la tarjeta de la landing). Antes era
      // "if isOwner()" a secas, así que cuando un barbero subía su foto el
      // write se denegaba, el error se tragaba un .catch y al recargar la
      // página volvía la inicial: la foto nunca se guardaba.
      //
      // OJO: "pd" NO va acá, aunque en tu versión sí estaba. "pd" es el
      // porcentaje que se queda el DUEÑO, y el barbero lo copiaba en cada
      // cobro (bsnap()), así que podía hacer, desde la consola del navegador:
      //
      //     updateDoc(doc(db, "staff", miUid), { pd: 0 })
      //
      // y a partir de ese momento todos sus cortes le quedaban íntegros: en
      // el reparto del dueño figuraba 0% y la plata no aparecía. El reparto se
      // cambia sólo desde Finanzas, que es una pantalla del dueño.
      allow update: if isOwner()
                    || (request.auth.uid == uid
                        && request.resource.data.diff(resource.data).affectedKeys()
                             .hasOnly(['n', 'foto', 'tel', 'ig', 'desc']));
    }

    // ---------- el perfil público de cada barbero, en su propio documento ----------
    //
    // Acá vive TODO lo que el barbero publica sobre sí: la foto, el nombre, el
    // oficio, la descripción, el teléfono, el Instagram y las redes. Se LEE sin
    // sesión (la landing tiene que mostrarlo al que no está logueado) y sólo lo
    // ESCRIBE el propio barbero, y sólo esos campos.
    //
    // POR QUÉ NO SE USA `diff(resource.data)` ACÁ. El bug que había:
    //
    //     allow create, update: if ... request.resource.data
    //                                  .diff(resource.data)
    //                                  .affectedKeys().hasOnly(['foto']);
    //
    // En un CREATE el documento no existe todavía, así que `resource` es null
    // y `resource.data` hace TRUENAR la regla: Firestore falla cerrada y
    // deniega. Comprobado contra las reglas publicadas: fotoBarbero/{uid} daba
    // 404 (no existía) y el setDoc de la app devolvía 403. Como además el
    // código se tragaba el error con un `.catch(() => {})`, la foto se guardaba
    // en users/ y staff/ y en la web no aparecía nunca: la tarjeta decía
    // "[Foto]".
    //
    // Acá no se menciona `resource` en ninguna rama, así que no hay forma de
    // que truene. Con `hasOnly` alcanza para acotar qué campos se pueden
    // escribir; no limita QUÉ se modifica de los que ya estaban, pero como la
    // lista blanca es exactamente la del perfil, escribir cualquier subconjunto
    // de esos campos es lo único que se puede llegar a hacer.
    match /fotoBarbero/{uid} {
      allow read: if true;
      allow create, update: if isSignedIn()
                            && request.auth.uid == uid
                            && request.resource.data.keys().hasOnly(
                                 ['foto', 'n', 'esp', 'desc', 'tel', 'ig', 'redes']);
      allow delete: if isSignedIn() && request.auth.uid == uid;
    }

    match /turnos/{id} {
      allow read: if isSignedIn() && (resource.data.uid == request.auth.uid || isBarberoOrOwner());
      // El "create" tiene que aceptar DOS casos distintos, y antes solo
      // aceptaba el primero:
      //
      //   1. El cliente reserva para si mismo: uid == su propio auth.uid.
      //   2. El barbero o el dueño cargan un turno desde el panel de Agenda
      //      a nombre de un cliente (cobro en la barbería). Ahí el `uid` del
      //      documento es el del CLIENTE, no el del barbero, así que la
      //      condición de arriba no se cumplía y el write se denegaba con
      //      "Missing or insufficient permissions": el barbero abría "+ Agendar
      //      turno", completaba todo y no lo guardaba nunca.
      //
      // Es la única colección que se había quedado sin la rama de staff;
      // /ocupados, /espera y /activos ya la tenían.
      //
      // Y además se fuerza el FORMATO de la fecha y de la hora (nada de HTML).
      // Con el formato cerrado, el dato que llega ya no puede ser marcado:
      // antes un cliente podía dejar
      //   "t": "<img src=x onerror=...>" y ese HTML se dibujaba crudo en la
      // Agenda del dueño y del barbero (probado con tests/xss.js: el payload
      // corría y window.__pwn quedaba en 1).
      allow create: if isSignedIn()
                    && (request.resource.data.uid == request.auth.uid || isBarberoOrOwner())
                    && request.resource.data.t.matches('^[0-2][0-9]:[0-5][0-9]$')
                    && request.resource.data.d.matches('^[0-9]{4}-[0-9]{2}-[0-9]{2}$');

      // El cliente puede mover o cancelar SU turno pendiente, y nada más.
      //
      // Antes (y en tu versión era): "if isBarberoOrOwner() || (resource.data
      // .uid == request.auth.uid)" para update Y delete, sin mirar qué campos
      // cambian ni en qué estado estaba. Eso le permitía a un cliente, desde
      // el SDK:
      //
      //   * marcarse el turno como "hecho" y ponerle el precio que quisiera
      //     (inflaba el Dashboard del dueño y el promedio por corte);
      //   * cambiar "precio", "pd", "hechoEn" o "bid" de un turno ya cobrado;
      //   * BORRAR un turno ya cobrado, y con él la evidencia de la venta.
      //
      // La segunda rama es para "borrar tus datos": el cliente anonimiza los
      // turnos que ya no están pendientes con { nm: "Cliente eliminado" }. Sin
      // ella no podría borrar su cuenta, porque la lista de arriba no incluía
      // "nm" y se caía el lote entero.
      allow update: if isBarberoOrOwner()
                    || (isSignedIn()
                        && resource.data.uid == request.auth.uid
                        && ((resource.data.st == 'pend'
                             && request.resource.data.st in ['pend', 'cancel']
                             && request.resource.data.diff(resource.data).affectedKeys()
                                  .hasOnly(['d', 't', 'bid', 'bl', 'st', 'mot', 'por', 'canEn']))
                            || (resource.data.st != 'pend'
                                && request.resource.data.diff(resource.data).affectedKeys()
                                     .hasOnly(['nm']))));
      // Borrar un turno es cosa de quien administra. Con "st": "cancel", que
      // queda registrado, el borrado físico queda arriba.
      allow delete: if isBarberoOrOwner();
    }

    match /ocupados/{id} {
      allow read: if true;
      allow create: if isSignedIn() && (request.resource.data.uid == request.auth.uid || isBarberoOrOwner());
      allow update, delete: if isSignedIn() && (resource.data.uid == request.auth.uid || isBarberoOrOwner());
    }

    match /activos/{id} {
      allow read, write: if isSignedIn() && (request.auth.uid == resource.data.uid || request.auth.uid == id.split('_')[0] || isBarberoOrOwner());
    }

    match /espera/{id} {
      allow read: if isSignedIn() && (resource.data.uid == request.auth.uid || isBarberoOrOwner());
      allow create: if isSignedIn() && (request.resource.data.uid == request.auth.uid || isBarberoOrOwner());
      allow update, delete: if isSignedIn() && (resource.data.uid == request.auth.uid || isBarberoOrOwner());
    }

    match /canjes/{id} {
      allow read: if isSignedIn() && (resource.data.uid == request.auth.uid || isBarberoOrOwner());
      allow create: if isSignedIn() && request.resource.data.uid == request.auth.uid;
      allow update, delete: if isBarberoOrOwner();
    }

    // Las ventas y los gastos son de un barbero concreto ("bid"). La UI ya
    // filtraba por "bid == me.uid", pero la regla no: con "read, write: if
    // isBarberoOrOwner()" un barbero podía, desde el SDK, borrar ventas del
    // dueño o cambiarle el precio a un cobro ajeno.
    //
    // OJO con la rama del dueño: "bsnap()" devuelve el "bid" del BARBERO al que
    // se le cobró (un id de tipo "b1"), no el uid del dueño. Por eso la rama
    // del dueño no lleva la comparación de "bid".
    match /ventas/{id} {
      allow read: if isOwner() || (isStaff() && resource.data.bid == request.auth.uid);
      allow create: if isOwner() || (isStaff() && request.resource.data.bid == request.auth.uid);
      allow update, delete: if isOwner() || (isStaff() && resource.data.bid == request.auth.uid);
    }

    // Los gastos los carga el dueño desde el negocio. No hay rama de barbero.
    match /gastos/{id} {
      allow read, create, update, delete: if isOwner();
    }
  }
}
